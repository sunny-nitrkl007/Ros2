cmake_minimum_required(VERSION 3.8)
project(lps_job_manager)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(lps_interfaces REQUIRED)
find_package(lps_core REQUIRED)
add_executable(lps_job_manager_node src/job_manager_node.cpp)
target_compile_features(lps_job_manager_node PRIVATE cxx_std_17)
ament_target_dependencies(lps_job_manager_node rclcpp lps_interfaces lps_core)
if(TARGET lps_pass_tracker_adapter)
  target_link_libraries(lps_job_manager_node lps_pass_tracker_adapter)
  target_compile_definitions(lps_job_manager_node PRIVATE LPS_HAS_PASS_TRACKER=1)
endif()
install(TARGETS lps_job_manager_node DESTINATION lib/${PROJECT_NAME})
ament_package()
